%----------RESEARCH----------
\section{Research Experience}
\entry{Institute for Quantum Computing (IQC), University of Waterloo}{May 2026 -- Aug. 2026}{Undergraduate Researcher (Mitacs Globalink Scholar), hosted by Michele Mosca and Vlad Gheorghiu}{Waterloo, ON, Canada}
\begin{itemize}\small
  \item Proved that Campbell and Howard's $6m+1$ construction for a layer of $m$ disjoint CCZ (Toffoli) gates is
  T-optimal among Hadamard-free circuits, via a lower bound computable in $O(n^3)$ time for every diagonal
  Clifford+T gate. Single-author, \href{https://arxiv.org/abs/2610.01024}{arXiv:2610.01024}; posters at QRE and UnitaryCon 2026.
  \item The bound certifies 193 of 311 phase blocks in PyZX/TODD optimizer output as T-optimal, against 113 for
  stabilizer nullity, and recovers the known optimum of 17 T gates for the $\mathrm{GF}(4)$ multiplier.
  \item Completed IQC's USEQIP program: quantum information coursework and 30+ hours of experiments on quantum hardware.
\end{itemize}

\entry{Magic State Labs}{Aug. 2026 -- Oct. 2026}{Research Consultant, with Pradeep Niroula}{Remote}
\begin{itemize}\small
  \item Built soundness rules for screening candidate problems for decoded quantum interferometry (DQI) and ran an
  AI-agent search under them that found two new candidate applications (\href{https://arxiv.org/abs/2610.06753}{arXiv:2610.06753}).
\end{itemize}

\entry{Caltech Institute for Quantum Information and Matter (IQIM)}{May 2025 -- Aug. 2025}{SURF Undergraduate Researcher, Preskill group, with Samson Wang and James Watson (Google Quantum AI)}{Pasadena, CA}
