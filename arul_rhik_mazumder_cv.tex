\section{Research Experience}
\entry{Institute for Quantum Computing (IQC), University of Waterloo}{May 2026 -- Aug. 2026}{Undergraduate Researcher (Mitacs Globalink Scholar), hosted by Michele Mosca and Vlad Gheorghiu}{Waterloo, ON, Canada}
\begin{itemize}\small
  \item Proved, in a \textbf{single-author} paper (\href{https://arxiv.org/abs/2610.01024}{arXiv:2610.01024}), that no circuit
  without internal Hadamard gates can implement a layer of $m$ non-overlapping Toffoli-type (CCZ) gates with fewer
  T gates than Campbell and Howard's $6m+1$ construction, using a lower bound, computable in polynomial time, for a
  broad class of diagonal gates.
  \item On the output of PyZX's TODD-based optimizer, the bound proves 193 of 311 circuit pieces already optimal,
  against 113 for stabilizer nullity, the standard prior bound.
\end{itemize}

\entry{Magic State Labs}{Aug. 2026 -- Oct. 2026}{Research Consultant, with Pradeep Niroula}{Remote}
\begin{itemize}\small
  \item Wrote criteria for checking whether a problem is a good fit for decoded quantum interferometry (DQI), a
  recent quantum optimization algorithm, and used them to steer an AI-agent search that found two new candidate
  applications (\href{https://arxiv.org/abs/2610.06753}{arXiv:2610.06753}).
\end{itemize}

\entry{Caltech Institute for Quantum Information and Matter (IQIM)}{May 2025 -- Aug. 2025}{SURF Undergraduate Researcher, Preskill group, with Samson Wang and James Watson (Google Quantum AI)}{Pasadena, CA}
\begin{itemize}\small
  \item Developed, as first author (\href{https://arxiv.org/abs/2608.13862}{arXiv:2608.13862}), a simulation method for early
  error-corrected hardware that runs many shallow random circuits and combines their results classically. For a given
  accuracy its circuits are exponentially shallower than standard Trotter simulation, at the cost of a modest increase
  in the number of runs.
  \item Proved that the method converges, bounded its cost, tightened the bounds for systems with weak interactions
  or conserved quantities, and applied them to ground-state energies and the response properties that chemists and
  materials scientists most want from a simulation.
\end{itemize}

\entry{BlueQubit}{Oct. 2025 -- Present}{Part-time Researcher, AI and Core teams}{Los Angeles, CA (remote)}
\begin{itemize}\small
  \item Led a six-person study, as first author, benchmarking GPU-accelerated quantum simulators against commercial and open-source platforms
  (\href{https://arxiv.org/abs/2607.09882}{arXiv:2607.09882}); the results fed into BlueQubit's simulator cost estimators.
\end{itemize}

\entry{Quantum Technologies Group, Carnegie Mellon University}{May 2023 -- Jun. 2025}{Researcher with Sridhar Tayur}{Pittsburgh, PA}
