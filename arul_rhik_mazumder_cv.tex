\section{Research Experience}
\entry{Institute for Quantum Computing (IQC), University of Waterloo}{May 2026 -- Aug. 2026}{Mitacs Globalink Research Intern, hosted by Michele Mosca and Vlad Gheorghiu}{Waterloo, ON, Canada}
\begin{itemize}\small
  \item Proved, in a \textbf{single-author} paper (\href{https://arxiv.org/abs/2610.01024}{arXiv:2610.01024}), that no circuit
  without internal Hadamard gates or measurement can implement a layer of $m$ non-overlapping Toffoli-type (CCZ) gates
  with fewer T gates than Campbell and Howard's $6m+1$ construction, which had been open since 2017. The bound is computable in polynomial time for a broad family of gates.
  \item On the output of PyZX's TODD-based optimizer, the bound proves 193 of 311 circuit pieces already optimal,
  against 113 for stabilizer nullity, the standard prior bound.
  \item Proved, in a second \textbf{single-author} paper (\href{https://arxiv.org/abs/2610.38189}{arXiv:2610.38189}), that randomized Toffoli gates need no extra qubits once
  the gate is large enough, using about 8 T gates per bit of precision. Also showed that oracles for random sets need
  exponentially many T gates while affine ones need a constant number.
\end{itemize}

\entry{Caltech Institute for Quantum Information and Matter (IQIM)}{May 2025 -- Aug. 2025}{SURF Fellow, Preskill group, with Samson Wang and James Watson (Google Quantum AI)}{Pasadena, CA}
\begin{itemize}\small
  \item Developed, as first author (\href{https://arxiv.org/abs/2608.13862}{arXiv:2608.13862}; \href{https://research.google/pubs/resource-efficient-quantum-matrix-processing-with-commutator-scaling/}{Google Research page}), a simulation method for early
  error-corrected hardware that runs many shallow random circuits and combines their results classically. For a given
  accuracy its circuits are exponentially shallower than standard Trotter simulation, at the cost of only a
  $(\log\log(1/\varepsilon))^2$ factor more runs.
  \item Proved that the method converges, bounded its cost, tightened the bounds for systems with weak interactions
  or conserved quantities, and applied them to ground-state energy and Green's function estimation.
\end{itemize}

\entry{Magic State Labs}{Aug. 2026 -- Present}{Research Consultant, with Pradeep Niroula}{Remote}
\begin{itemize}\small
  \item Developed screening checks, formally verified by Pradeep Niroula, that test in seconds whether a problem suits decoded quantum interferometry (DQI); showed DQI beats a simple classical baseline by at most a $(\sqrt{2}-1)/2 \approx 21\%$ fraction of constraints under worst-case decoding, and found two new candidate problems that escape it (\href{https://arxiv.org/abs/2610.06753}{arXiv:2610.06753}).
\end{itemize}
